# Antminer S19J v2.5 - legacy NAND U-Boot SD boot configuration
# Requires FAT partition 1 containing zImage and am335x-antminer.dtb.
# Verify MMC index and SD boot support on the serial console before use.
bootfile=zImage
fdtfile=am335x-antminer.dtb
loadaddr=0x80008000
fdtaddr=0x88000000
mmcargs=setenv bootargs console=ttyS0,115200n8 root=/dev/mmcblk0p2 rw rootwait
uenvcmd=run mmcargs; if fatload mmc 0:1 ${loadaddr} ${bootfile}; then if fatload mmc 0:1 ${fdtaddr} ${fdtfile}; then bootz ${loadaddr} - ${fdtaddr}; fi; fi
